## Try it

<BasicDemo client:visible />

```tsx
import { createTour, Tour } from '@awesome-tour/react';

const tour = createTour({
  id: 'onboarding',
  steps: [
    { id: 'welcome', title: 'Welcome to Acme 👋', description: 'A step without a target is centered.' },
    { id: 'nav', target: '#nav', side: 'right', title: 'Navigation' },
    { id: 'search', target: '#search', title: 'Search' },
    { id: 'stats', target: ['#revenue', '#churn'], title: 'Multiple targets' },
  ],
});

export function Onboarding() {
  return (
    <Tour.Root tour={tour}>
      <Tour.Trigger>Start tour</Tour.Trigger>
      <Tour.Portal>
        <Tour.Overlay />
        <Tour.Content className="card">
          <Tour.Arrow />
          <Tour.Progress />
          <Tour.Title />
          <Tour.Description />
          <Tour.Prev>Back</Tour.Prev>
          <Tour.Next>Next</Tour.Next>
        </Tour.Content>
      </Tour.Portal>
    </Tour.Root>
  );
}
```
